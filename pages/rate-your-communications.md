<script>
(function(){
  var current = 1, total = 10, scores = {};
  var answers = {};

  function showQ(n) {
    document.querySelectorAll(".ka-question").forEach(function(q){ q.classList.remove("active"); });
    var el = document.querySelector('[data-q="'+n+'"]');
    if (el) el.classList.add("active");
    // a11y: move focus to the new question so it is announced (WCAG 2.4.3 / 4.1.3)
    if (el) { var h = el.querySelector(".ka-q-title"); if (h) { h.setAttribute("tabindex", "-1"); h.focus(); } }
    document.getElementById("ka-q-num").textContent = Math.min(n, total);
    document.getElementById("ka-progress").style.width = Math.round((n/11)*100)+"%";
    document.getElementById("ka-prev").style.display = n > 1 ? "inline-block" : "none";
    var isEmail = n === 11;
    document.getElementById("ka-next").style.display = isEmail ? "none" : "inline-block";
    document.getElementById("ka-submit").style.display = isEmail ? "inline-block" : "none";
    document.getElementById("ka-quiz-intro").style.display = isEmail ? "none" : "block";
  }

  document.getElementById("ka-next").addEventListener("click", function(){
    if (current <= total) {
      var sel = document.querySelector('input[name="q'+current+'"]:checked');
      var qErr = document.getElementById("ka-q-err");
      if (!sel) {
        qErr.textContent = "Please choose one answer to continue.";
        qErr.hidden = false;
        var first = document.querySelector('input[name="q'+current+'"]');
        if (first) first.focus();
        return;
      }
      qErr.hidden = true;
      scores["q"+current] = parseInt(sel.value);
    }
    current++;
    showQ(current);
  });

  document.getElementById("ka-prev").addEventListener("click", function(){
    if (current > 1) { current--; showQ(current); }
  });

  // Auto-advance on selection
  document.querySelectorAll(".ka-answers input").forEach(function(inp){
    inp.addEventListener("change", function(){
      var labels = this.closest(".ka-answers").querySelectorAll("label");
      labels.forEach(function(l){ l.classList.remove("selected"); });
      this.closest("label").classList.add("selected");
      answers[this.name] = this.closest("label").textContent.trim();
    });
  });

  document.getElementById("ka-submit").addEventListener("click", function(){
    var name = document.getElementById("ka-name").value.trim();
    var email = document.getElementById("ka-email").value.trim();
    var gateErr = document.getElementById("ka-gate-err"), bad = [];
    [["ka-name", name, "your name"], ["ka-email", email, "your email address"]].forEach(function(f){
      var inp = document.getElementById(f[0]);
      inp.setAttribute("aria-invalid", f[1] ? "false" : "true");
      if (!f[1]) bad.push(f);
    });
    if (bad.length) {
      gateErr.textContent = "Please enter " + bad.map(function(b){ return b[2]; }).join(" and ") + " to see your results.";
      gateErr.hidden = false;
      document.getElementById(bad[0][0]).focus();
      return;
    }
    gateErr.hidden = true;

    var total_score = 0;
    for (var k in scores) { total_score += scores[k]; }

    // Send to server (fire and forget)
    fetch(kaQuizConfig.ajaxUrl, {
      method: "POST",
      headers: {"Content-Type":"application/x-www-form-urlencoded"},
      body: "action=ka_quiz_submit&name="+encodeURIComponent(name)+"&email="+encodeURIComponent(email)+"&church="+encodeURIComponent(document.getElementById("ka-church").value)+"&score="+total_score+"&newsletter="+(document.getElementById("ka-newsletter").checked?"1":"")+"&nonce="+kaQuizConfig.nonce+"&answers="+encodeURIComponent(JSON.stringify(answers))
    });

    document.getElementById("ka-quiz-form").style.display = "none";
    document.getElementById("ka-quiz-intro").style.display = "none";
    document.getElementById("ka-results").style.display = "block";

    var result_id, score_el, bar_el;
    if (total_score <= 39) {
      result_id = "ka-result-flying-blind"; score_el = "ka-score-display"; bar_el = "ka-score-bar-fill";
    } else if (total_score <= 69) {
      result_id = "ka-result-momentum"; score_el = "ka-score-display-2"; bar_el = "ka-score-bar-fill-2";
    } else {
      result_id = "ka-result-almost"; score_el = "ka-score-display-3"; bar_el = "ka-score-bar-fill-3";
    }
    document.getElementById(result_id).style.display = "block";
    document.getElementById(score_el).textContent = total_score;
    // a11y: announce the result by moving focus to its heading
    var rh = document.querySelector("#" + result_id + " h2");
    if (rh) { rh.setAttribute("tabindex", "-1"); rh.focus(); }
    setTimeout(function(){ document.getElementById(bar_el).style.width = total_score+"%"; }, 100);
  });

  showQ(1);
})();
</script>
